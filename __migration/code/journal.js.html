 async function load(){

  try{
   const [journal,feed]=await Promise.allSettled([fetchJson('/api/journal'),fetchJson('/api/news')]);
   let changed=initial||dirty;
   if(journal.status==='fulfilled'){changed||=JSON.stringify(authored)!==JSON.stringify(journal.value.items);authored=journal.value.items;}
   newsFailed=feed.status!=='fulfilled';
   if(!newsFailed){const signature=JSON.stringify(feed.value.items);changed||=signature!==lastNews;official=feed.value.items;lastNews=signature;checkedAt=feed.value.checkedAt;}
   merge();
   // Do not replace a focused card or interrupt an open article during polling.
   if(changed&&!dialog.open&&!newsList.contains(document.activeElement)&&!list.contains(document.activeElement)){render();dirty=false;}else{dirty=changed;refreshCheck();}
   if(journal.status!=='fulfilled')status.textContent='As publicações autorais estão indisponíveis no momento.';
   if(newsFailed&&!official.length){newsStatus.textContent='Notícias indisponíveis no momento.';newsList.replaceChildren(make('p','','Não foi possível consultar as notícias. Uma nova tentativa será feita em instantes.'));}
   if(initial){const item=items.find(i=>i.id===new URL(location.href).searchParams.get('artigo'));if(item)open(item);initial=false;}
  }finally{loading=false;}
