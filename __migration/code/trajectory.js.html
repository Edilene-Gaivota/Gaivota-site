(() => {
  const drawer=document.querySelector('#trajectory-drawer'),toggle=document.querySelector('#trajectory-toggle'),panel=document.querySelector('#trajectory-panel');
  if(!drawer||!toggle||!panel)return;
  const motion=matchMedia('(prefers-reduced-motion:reduce)'),view=document.querySelector('#drawer-review'),controls=document.querySelector('#drawer-review-controls'),pause=document.querySelector('#review-pause');
  let opened=false,items=[],index=0,paused=motion.matches,hovering=false,visible=false,rotation,poll,loading=false;
  function render(){view.replaceChildren();controls.hidden=items.length<2;if(!items.length){const p=document.createElement('p');p.className='drawer-review-state';p.textContent='As avaliações autorizadas para exibição aparecerão aqui.';view.append(p);return;}
    index=(index+items.length)%items.length;const item=items[index],quote=document.createElement('blockquote');if(item.comment){const p=document.createElement('p');p.textContent=item.comment;quote.append(p);}if(item.choices.length){const list=document.createElement('ul');item.choices.forEach(text=>{const li=document.createElement('li');li.textContent=text;list.append(li);});quote.append(list);}const cite=document.createElement('cite');cite.textContent=item.name||'Avaliação anônima';quote.append(cite);view.append(quote);document.querySelector('#review-position').textContent=`${index+1} de ${items.length}`;}
  function updatePause(){pause.textContent=paused?'Reproduzir':'Pausar';pause.setAttribute('aria-pressed',String(paused));}
  async function load(){if(loading)return;loading=true;try{const response=await fetch('/api/feedback',{cache:'no-store'});if(!response.ok)throw new Error();const data=await response.json(),current=items[index]?.id;items=Array.isArray(data.items)?data.items:[];index=Math.max(0,items.findIndex(item=>item.id===current));render();}catch{if(!items.length){view.replaceChildren();const p=document.createElement('p');p.className='drawer-review-state';p.textContent='Não foi possível carregar as avaliações agora. Tente reabrir este espaço em alguns instantes.';view.append(p);}}finally{loading=false;}}
  function timers(){clearInterval(rotation);clearInterval(poll);if(!opened)return;rotation=setInterval(()=>{if(!paused&&!motion.matches&&!hovering&&visible&&!document.hidden&&!panel.querySelector('.drawer-feedback').contains(document.activeElement)&&items.length>1){index++;render();}},7000);poll=setInterval(()=>{if(visible&&!document.hidden)load();},60000);}
  function setOpen(value,restore=false){opened=value;drawer.classList.toggle('is-open',opened);panel.inert=!opened;panel.setAttribute('aria-hidden',String(!opened));toggle.setAttribute('aria-expanded',String(opened));toggle.setAttribute('aria-label',opened?'Fechar feedbacks':'Abrir feedbacks');if(opened)load();timers();if(restore)toggle.focus({preventScroll:true});}
  toggle.addEventListener('click',()=>setOpen(!opened));document.querySelector('#trajectory-close').addEventListener('click',()=>setOpen(false,true));
  drawer.addEventListener('keydown',event=>{if(event.key==='Escape'&&opened){event.preventDefault();setOpen(false,true);}});
  panel.querySelector('.drawer-feedback').addEventListener('pointerenter',()=>hovering=true);panel.querySelector('.drawer-feedback').addEventListener('pointerleave',()=>hovering=false);
  for(const [id,direction] of [['#review-prev',-1],['#review-next',1]])document.querySelector(id).addEventListener('click',()=>{if(!items.length)return;index+=direction;render();document.querySelector('#review-announcement').textContent=`Avaliação ${index+1} de ${items.length}`;});
  pause.addEventListener('click',()=>{paused=!paused;updatePause();});motion.addEventListener('change',()=>{paused=motion.matches;updatePause();});
  new IntersectionObserver(entries=>{visible=entries[0].isIntersecting;},{threshold:.1}).observe(panel);
  document.addEventListener('feedback:saved',()=>{index=0;if(opened)load();});updatePause();setOpen(false);
})();
